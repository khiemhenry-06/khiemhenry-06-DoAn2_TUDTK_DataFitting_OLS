\section{Phần 1: Lý Thuyết Data Fitting và Minh Họa}

\subsection{Bài Toán Data Fitting}

\subsubsection{Phát biểu bài toán tổng quát}

Cho tập dữ liệu $\mathcal{D} = \{(\mathbf{x}_i, y_i)\}_{i=1}^{n}$ với $\mathbf{x}_i \in \mathbb{R}^p$, $y_i \in \mathbb{R}$. Bài toán data fitting là tìm hàm $f: \mathbb{R}^p \to \mathbb{R}$ trong một lớp hàm cho trước sao cho $f$ xấp xỉ tốt nhất ánh xạ từ $\mathbf{x}_i$ đến $y_i$ theo một tiêu chí đã định.

Trong mô hình hồi quy tuyến tính, ta giả thiết:
\begin{equation}
    y_i = \beta_0 + \beta_1 x_{i1} + \beta_2 x_{i2} + \cdots + \beta_p x_{ip} + \varepsilon_i = \mathbf{x}_i^T \boldsymbol{\beta} + \varepsilon_i
\end{equation}
với $\boldsymbol{\beta} = (\beta_0, \beta_1, \ldots, \beta_p)^T \in \mathbb{R}^{p+1}$ là vector tham số cần ước lượng và $\varepsilon_i$ là nhiễu ngẫu nhiên.

Viết dưới dạng ma trận với $\mathbf{X} \in \mathbb{R}^{n \times (p+1)}$ (ma trận design có cột đầu toàn 1):
\begin{equation}
    \mathbf{y} = \mathbf{X}\boldsymbol{\beta} + \boldsymbol{\varepsilon}
\end{equation}

\subsubsection{Các Giả Thiết Gauss--Markov}

Các giả thiết Gauss--Markov (GM1 -- GM5) là nền tảng cho việc chứng minh tính tối ưu của ước lượng OLS:

\begin{itemize}
    \item \textbf{GM1. Tuyến tính:} $\mathbf{y} = \mathbf{X}\boldsymbol{\beta} + \boldsymbol{\varepsilon}$.
    \item \textbf{GM2. Không hoàn hảo đa cộng tuyến:} $\text{rank}(\mathbf{X}) = p + 1$ (các cột độc lập tuyến tính).
    \item \textbf{GM3. Ngoại sinh:} $E[\boldsymbol{\varepsilon} | \mathbf{X}] = \mathbf{0}$, tức $E[\varepsilon_i | \mathbf{x}_i] = 0$.
    \item \textbf{GM4. Đồng phương sai:} $\text{Var}(\boldsymbol{\varepsilon} | \mathbf{X}) = \sigma^2 \mathbf{I}_n$, tức $\text{Var}(\varepsilon_i) = \sigma^2$ và $\text{Cov}(\varepsilon_i, \varepsilon_j) = 0$ với $i \neq j$.
    \item \textbf{GM5. Phần dư Chuẩn:} $\boldsymbol{\varepsilon} | \mathbf{X} \sim \mathcal{N}(\mathbf{0}, \sigma^2 \mathbf{I}_n)$.
\end{itemize}

\subsection{Phương Pháp Ordinary Least Squares (OLS)}

\subsubsection{Hàm mất mát và nghiệm OLS}

OLS tìm $\hat{\boldsymbol{\beta}}$ tối thiểu hóa tổng bình phương phần dư (Residual Sum of Squares):
\begin{equation}
    \text{RSS}(\boldsymbol{\beta}) = \|\mathbf{y} - \mathbf{X}\boldsymbol{\beta}\|_2^2 = \sum_{i=1}^{n}(y_i - \mathbf{x}_i^T \boldsymbol{\beta})^2
\end{equation}

\textbf{Định lý (Nghiệm OLS --- Normal Equations).} Nếu $\mathbf{X}^T\mathbf{X}$ khả nghịch, nghiệm OLS duy nhất là:
\begin{equation}
    \hat{\boldsymbol{\beta}}_{\text{OLS}} = (\mathbf{X}^T\mathbf{X})^{-1}\mathbf{X}^T\mathbf{y}
    \label{eq:ols}
\end{equation}

\textit{Chứng minh.} Tính đạo hàm theo $\boldsymbol{\beta}$ và cho bằng không:
\[
    \nabla_{\boldsymbol{\beta}} \text{RSS} = -2\mathbf{X}^T(\mathbf{y} - \mathbf{X}\boldsymbol{\beta}) = \mathbf{0}
\]
Giải ra: $\mathbf{X}^T\mathbf{X}\boldsymbol{\beta} = \mathbf{X}^T\mathbf{y}$, từ đó suy ra công thức (\ref{eq:ols}).

\textbf{Cài đặt Python.} Hàm \texttt{ols\_fit(X, y)} trong file \texttt{part1/ols\_implementation.py} cài đặt công thức trên hoàn toàn bằng NumPy, sử dụng \texttt{np.linalg.pinv} để đảm bảo tính ổn định số:

\begin{lstlisting}[caption={Trích đoạn hàm ols\_fit}]
XtX = X_design.T @ X_design
XtX_inv = np.linalg.pinv(XtX)
beta = XtX_inv @ X_design.T @ y_np
y_hat = X_design @ beta
residuals = y_np - y_hat
\end{lstlisting}

Kết quả trả về bao gồm: $\hat{\boldsymbol{\beta}}$, $\hat{\mathbf{y}}$, phần dư, RSS, TSS, $R^2$, $\hat{\sigma}^2$, ma trận hiệp phương sai, sai số chuẩn, thống kê $t$ và $p$-value.

\subsubsection{Ma Trận Chiếu và Hat Matrix}

\textbf{Định nghĩa.} Ma trận chiếu (projection matrix hay hat matrix) là:
\begin{equation}
    \mathbf{H} = \mathbf{X}(\mathbf{X}^T\mathbf{X})^{-1}\mathbf{X}^T \in \mathbb{R}^{n \times n}
    \label{eq:hat}
\end{equation}

\textbf{Tính chất của H:}
\begin{enumerate}
    \item $\mathbf{H}^2 = \mathbf{H}$ (idempotent).
    \item $\mathbf{H}^T = \mathbf{H}$ (đối xứng).
    \item Giá trị riêng của $\mathbf{H}$: chỉ là 0 hoặc 1.
    \item $\text{rank}(\mathbf{H}) = p + 1$.
    \item Giá trị fitted: $\hat{\mathbf{y}} = \mathbf{H}\mathbf{y}$; phần dư: $\hat{\boldsymbol{\varepsilon}} = (\mathbf{I} - \mathbf{H})\mathbf{y}$.
\end{enumerate}

\textbf{Cài đặt và kiểm chứng.} Hàm \texttt{hat\_matrix(X)} tính $\mathbf{H}$ theo công thức (\ref{eq:hat}) và hàm \texttt{check\_idempotent(H)} kiểm tra $\mathbf{H}^2 \approx \mathbf{H}$ trong sai số $10^{-8}$. Unit test xác nhận ma trận Hat luôn có tính idempotent trên dữ liệu giả lập.

\subsection{Đánh Giá Mô Hình}

\subsubsection{Hệ số xác định $R^2$ và $R^2$ hiệu chỉnh}

\begin{equation}
    R^2 = 1 - \frac{\text{RSS}}{\text{TSS}} = 1 - \frac{\sum_{i=1}^{n}(y_i - \hat{y}_i)^2}{\sum_{i=1}^{n}(y_i - \bar{y})^2}
\end{equation}

$R^2$ hiệu chỉnh phạt cho việc thêm biến không cần thiết:
\begin{equation}
    \bar{R}^2 = 1 - (1 - R^2)\frac{n-1}{n-p}
\end{equation}

\subsubsection{Kiểm Định Giả Thuyết}

\textbf{Kiểm định $t$ cho từng hệ số.} Với giả thuyết $H_0: \beta_j = 0$:
\begin{equation}
    t_j = \frac{\hat{\beta}_j}{\text{se}(\hat{\beta}_j)}, \quad \text{se}(\hat{\beta}_j) = \sqrt{\hat{\sigma}^2 [(\mathbf{X}^T\mathbf{X})^{-1}]_{jj}}
\end{equation}

Dưới $H_0$, $t_j \sim t_{n-p}$. Giá trị $p$-value $= 2P(|T| > |t_j|)$.

\textbf{Kiểm định $F$ tổng thể.} Kiểm tra ý nghĩa của toàn bộ mô hình:
\begin{equation}
    F = \frac{(\text{TSS} - \text{RSS})/(p-1)}{\text{RSS}/(n-p)} \sim F_{p-1,\, n-p}
\end{equation}

Hàm \texttt{model\_metrics(y, y\_hat, p)} và \texttt{coef\_inference(X, y, beta\_hat, sigma2)} cài đặt đầy đủ các kiểm định trên, trả về DataFrame gồm $\hat{\beta}$, se, $t$, $p$-value và khoảng tin cậy 95\%.

\subsection{Các Vấn Đề Nâng Cao trong Data Fitting}

\subsubsection{Đa cộng tuyến (Multicollinearity)}

Khi các biến giải thích tương quan mạnh, ma trận $\mathbf{X}^T\mathbf{X}$ gần suy biến, dẫn đến phương sai của $\hat{\boldsymbol{\beta}}$ tăng vọt.

\textbf{Variance Inflation Factor (VIF):}
\begin{equation}
    \text{VIF}_j = \frac{1}{1 - R_j^2}
\end{equation}
trong đó $R_j^2$ là $R^2$ khi hồi quy $X_j$ trên các biến còn lại. Nếu $\text{VIF}_j > 10$, biến $j$ có đa cộng tuyến nghiêm trọng.

Hàm \texttt{vif(X)} cài đặt tính VIF cho từng biến bằng cách lặp hồi quy OLS trên các tập biến phụ.

\subsubsection{Hồi Quy Ridge và Lasso (Regularization)}

\textbf{Ridge Regression} thêm hệ phạt $L_2$ vào hàm mất mát:
\begin{equation}
    \hat{\boldsymbol{\beta}}_{\text{Ridge}} = \arg\min_{\boldsymbol{\beta}} \left\{ \|\mathbf{y} - \mathbf{X}\boldsymbol{\beta}\|_2^2 + \lambda\|\boldsymbol{\beta}\|_2^2 \right\} = (\mathbf{X}^T\mathbf{X} + \lambda\mathbf{I})^{-1}\mathbf{X}^T\mathbf{y}
    \label{eq:ridge}
\end{equation}

\textbf{Lasso Regression} thêm hệ phạt $L_1$:
\begin{equation}
    \hat{\boldsymbol{\beta}}_{\text{Lasso}} = \arg\min_{\boldsymbol{\beta}} \left\{ \|\mathbf{y} - \mathbf{X}\boldsymbol{\beta}\|_2^2 + \lambda\|\boldsymbol{\beta}\|_1 \right\}
\end{equation}

Lasso có khả năng đưa hệ số về đúng 0 (chọn biến tự động --- sparsity). Ridge chỉ co nhỏ hệ số nhưng không triệt tiêu.

\textbf{Ridge Trace.} Vẽ giá trị $\hat{\beta}_j$ theo $\lambda$ trên trục log giúp quan sát sự co nhỏ dần của các hệ số khi $\lambda$ tăng.

\subsubsection{Phân Tích Phần Dư (Residual Analysis)}

Phân tích phần dư là bước chẩn đoán quan trọng để kiểm tra các giả thiết của mô hình hồi quy. Bốn biểu đồ chẩn đoán tiêu chuẩn bao gồm:

\begin{enumerate}
    \item \textbf{Residuals vs Fitted:} Kiểm tra tính tuyến tính và đồng phương sai. Nếu mẫu hình phễu (funnel shape) xuất hiện, cho thấy phương sai không đồng nhất.
    \item \textbf{Q-Q Plot:} Kiểm tra tính chuẩn của phần dư. Nếu các điểm nằm sát đường $y = x$, phần dư tuân theo phân phối chuẩn.
    \item \textbf{Scale-Location:} Vẽ $\sqrt{|e_i^*|}$ theo $\hat{y}_i$ (với $e_i^*$ là phần dư chuẩn hóa). Kiểm tra tính đồng phương sai (homoscedasticity).
    \item \textbf{Cook's Distance:} Đo mức ảnh hưởng của từng quan sát lên mô hình:
    \begin{equation}
        D_i = \frac{e_i^2}{p \cdot \hat{\sigma}^2} \cdot \frac{h_{ii}}{(1 - h_{ii})^2}
    \end{equation}
    trong đó $h_{ii}$ là phần tử đường chéo thứ $i$ của ma trận Hat. Giá trị $D_i > 4/n$ thường được xem là có ảnh hưởng lớn.
\end{enumerate}

\subsubsection{Cross-Validation và Lựa Chọn Mô Hình}

\textbf{$k$-Fold Cross-Validation:} Chia dữ liệu thành $k$ phần bằng nhau. Mỗi lần dùng $k-1$ phần để huấn luyện, 1 phần để kiểm tra. Lặp $k$ lần và lấy trung bình:
\begin{equation}
    \text{CV}(k) = \frac{1}{k}\sum_{i=1}^{k}\text{MSE}_i
\end{equation}

Cross-validation được dùng để chọn siêu tham số $\lambda$ tối ưu cho Ridge/Lasso mà không bị overfitting trên tập train.

\textbf{Tiêu chí lựa chọn mô hình:}
\begin{equation}
    \text{AIC} = n\ln\left(\frac{\text{RSS}}{n}\right) + 2(p+2), \quad
    \text{BIC} = n\ln\left(\frac{\text{RSS}}{n}\right) + (p+2)\ln n
\end{equation}

\subsection{Minh Họa Định Lý Gauss--Markov bằng Mô Phỏng Monte Carlo}

\subsubsection{Mục tiêu và Thiết lập mô phỏng}

Mục tiêu của phần này là thực nghiệm tính không chệch (Unbiased) của ước lượng OLS thông qua mô phỏng trên tập dữ liệu giả lập. Nhóm thiết lập một mô hình thực tế có dạng:
\begin{equation}
    y = \mathbf{x}^T \boldsymbol{\beta} + \varepsilon
\end{equation}
Trong đó, vector tham số thực được ấn định là $\boldsymbol{\beta} = (2.5, 1.5, -3.0)^T$. Quá trình mô phỏng được thực hiện qua $M = 1000$ lần lặp. Trong mỗi lần lặp, một nhiễu trắng $\varepsilon$ được sinh ra theo phân phối chuẩn $\mathcal{N}(0,1)$ và cộng vào mô hình để tạo ra tập dữ liệu $y$ mới.

\subsubsection{Phân tích kết quả thực nghiệm}

Dựa trên kết quả chạy thuật toán, nhóm thu được phân phối của các ước lượng $\hat{\beta}_1$ qua 1000 lần thử nghiệm.

\end{itemize}

\subsection{Tiền Xử Lý Dữ Liệu --- DataPipeline}

Class \texttt{DataPipeline} (file \texttt{part2/data\_pipeline.py}) tự động hóa toàn bộ quá trình tiền xử lý với thiết kế chống rò rỉ dữ liệu (data leakage prevention):

\begin{enumerate}
    \item \textbf{Xử lý ``tbd'':} Tự tìm chuỗi \texttt{tbd} ở cột \texttt{User\_Score}, chuyển sang \texttt{NaN}, ép kiểu \texttt{float}.
    \item \textbf{Điền khuyết (Imputation):} Dùng \textbf{Median} tính từ tập Train (phương pháp MV2 --- robust với outliers).
    \item \textbf{Rút gọn Publisher:} Giữ top 30 nhà phát hành phổ biến nhất, gom nhóm còn lại thành ``Other''.
    \item \textbf{One-Hot Encoding:} Mã hóa biến phân loại \texttt{Platform}, \texttt{Genre}, \texttt{Publisher}.
    \item \textbf{Chuẩn hóa Z-score:} $x_j^{\text{std}} = (x_j - \bar{x}_j) / s_j$ cho mọi biến số.
    \item \textbf{Căn chỉnh chiều:} Đảm bảo tập Test có cùng số cột và thứ tự cột với tập Train.
\end{enumerate}

\textbf{Quy trình sử dụng:}
\begin{lstlisting}[caption={Sử dụng DataPipeline đúng chuẩn}]
pipeline = DataPipeline(top_n_publishers=30)
X_train_clean = pipeline.fit_transform(X_train)  # Fit + Transform
X_test_clean = pipeline.transform(X_test)         # Chi Transform
\end{lstlisting}

Phương thức \texttt{fit()} học các thống kê (median, mean, std, danh sách cột) từ tập Train, còn \texttt{transform()} chỉ áp dụng các giá trị đã học --- ngăn chặn data leakage hoàn toàn.

\subsection{Xây Dựng và So Sánh Mô Hình}

Ba mô hình bắt buộc được xây dựng trong file \texttt{part2/model\_comparison.py}:

\subsubsection{Mô hình 1: OLS Cơ bản (Full Model)}

Hồi quy OLS với tất cả các biến sau tiền xử lý, sử dụng \texttt{statsmodels.OLS}. Đây là baseline để so sánh.

\subsubsection{Mô hình 2: OLS Chọn Biến (Stepwise Backward)}

Thuật toán loại bỏ dần (backward elimination) tự động loại các biến có $p$-value $> 0.05$:

\begin{lstlisting}[caption={Backward Elimination}]
def backward_elimination(X, y, significance_level=0.05):
    features = X.columns.tolist()
    while len(features) > 0:
        model = sm.OLS(y, sm.add_constant(X[features])).fit()
        p_values = model.pvalues.drop('const', errors='ignore')
        if p_values.max() > significance_level:
            features.remove(p_values.idxmax())
        else:
            break
    return sm.OLS(y, sm.add_constant(X[features])).fit(), features
\end{lstlisting}

\subsubsection{Mô hình 3: Ridge Regression (với Cross-Validation)}

Sử dụng \texttt{RidgeCV} từ scikit-learn với $k = 5$ fold, thử nghiệm các giá trị $\alpha \in \{0.01, 0.1, 1.0, 10.0, 100.0, 1000.0\}$. Mô hình tự động chọn $\alpha$ tối ưu qua cross-validation.

\subsection{Đánh Giá Mô Hình trên Tập Test}

\subsubsection{Mục tiêu và Phương pháp}

Mục tiêu của chương này là định lượng và so sánh hiệu suất dự báo của ba mô hình hồi quy đã được huấn luyện trong Phần 2 của đồ án. Toàn bộ quá trình đánh giá được thực hiện trên tập kiểm tra (test set) chiếm 20\% dữ liệu ($n_{\text{test}} = 3344$), được tách ra hoàn toàn trước bước huấn luyện nhằm đảm bảo tính khách quan.

Ba chỉ số đo lường được sử dụng theo yêu cầu của đề bài:
\begin{itemize}
    \item \textbf{MAE (Mean Absolute Error):} Sai số tuyệt đối trung bình, đơn vị triệu USD. Phản ánh trực quan mức độ lệch của dự báo so với giá trị thực.
    \item \textbf{RMSE (Root Mean Squared Error):} Căn bậc hai trung bình bình phương sai số. Phạt nặng hơn với các sai số lớn, phù hợp để phát hiện dự báo cực kỳ lệch.
    \item \textbf{R² (Hệ số xác định):} Tỷ lệ phần trăm biến động của \texttt{Global\_Sales} được mô hình giải thích. Giá trị càng gần 1.0 thì mô hình càng tốt.
\end{itemize}

\end{itemize}

File \texttt{part2/advanced\_methods.py} (Đỗ Trung Kiên thực hiện) tiến hành phân tích phần dư đầy đủ cho mô hình này:

\subsubsection{Biểu đồ 1 --- Residuals vs Fitted}
Biểu đồ này vẽ phần dư ($\text{residuals} = y_{\text{thực}} - \hat{y}_{\text{dự báo}}$) theo giá trị dự đoán của mô hình. Các điểm dữ liệu phân tán tương đối ngẫu nhiên và đối xứng quanh đường ngang $y = 0$, không biểu hiện xu hướng phi tuyến rõ ràng. Đường làm trơn LOWESS (đường cam) gần như nằm ngang ở vùng giá trị dự đoán thấp và trung bình, xác nhận mối quan hệ tuyến tính cơ bản giữa các đặc trưng và \texttt{Global\_Sales}.

Tuy nhiên, ở vùng giá trị dự đoán lớn ($\text{fitted values} > 3$), phần dư có xu hướng dương và phân tán rộng hơn, gợi ý sự tồn tại của heteroscedasticity --- phương sai phần dư tăng theo giá trị dự đoán. Đây là đặc trưng phổ biến trong dữ liệu doanh thu thực tế có phân phối lệch phải.

\subsubsection{Biểu đồ 2 --- Normal Q-Q Plot}
Biểu đồ Q-Q so sánh phân phối thực nghiệm của phần dư chuẩn hóa với phân phối chuẩn lý thuyết. Phần lớn các điểm nằm sát đường đỏ lý thuyết ở vùng trung tâm ($-2$ đến $+2$), xác nhận giả định phân phối chuẩn tương đối được thỏa mãn với khối dữ liệu chính.

Phần đuôi phải lệch mạnh khỏi đường chuẩn --- đây là hệ quả trực tiếp từ đặc trưng phân phối lệch phải (right-skewed) của \texttt{Global\_Sales}, nơi tồn tại các tựa game bom tấn có doanh thu cực cao như Wii Sports (82.53 triệu USD). Kết quả kiểm định Shapiro-Wilk ($p < 0.05$) cũng phù hợp với quan sát trực quan này. Để cải thiện, phép biến đổi $\log(\text{Global\_Sales})$ được gợi ý như một hướng xử lý trong tương lai.

\subsubsection{Biểu đồ 3 --- Scale-Location}
Biểu đồ Scale-Location vẽ căn bậc hai của giá trị tuyệt đối phần dư chuẩn hóa theo giá trị dự đoán, nhằm kiểm tra tính đồng đều của phương sai (homoscedasticity). Đường làm trơn LOWESS có xu hướng dốc lên rõ rệt, xác nhận hiện tượng heteroscedasticity: phương sai phần dư tăng khi giá trị dự đoán tăng.

Phát hiện này nhất quán với kết quả biểu đồ 1 và phản ánh cấu trúc thực tế của thị trường game --- các tựa game có doanh thu cao thường có độ biến động lớn hơn nhiều so với các tựa game bình thường. Để khắc phục, có thể áp dụng Weighted Least Squares (WLS) hoặc biến đổi logarit biến mục tiêu trước khi huấn luyện.

\subsubsection{Biểu đồ 4 --- Leverage và Cook's Distance}
Biểu đồ này kết hợp Leverage (mức độ ``đặc biệt'' của một điểm trong không gian đặc trưng) và Cook's Distance (tổng ảnh hưởng của điểm đó lên vector hệ số) để phát hiện các quan sát influential. Phân tích xác định được 315 điểm có Cook's Distance vượt ngưỡng $4/n$.

Các điểm outlier tập trung ở vùng Standardized Residuals lớn ($> 5$) với Leverage thấp, tức là chúng có phần dư lớn nhưng không gây ảnh hưởng mạnh đến ước lượng hệ số. Đây thường là các tựa game cực kỳ nổi tiếng (Nintendo blockbusters) có doanh thu thực tế vượt xa dự đoán của mô hình. Nhóm giữ nguyên các điểm này trong phân tích vì chúng là các quan sát hợp lệ, phản ánh thực tế thị trường chứ không phải lỗi dữ liệu.

Nintendo là nhà phát hành có doanh thu trung bình vượt trội nhờ các thương hiệu nhượng quyền kinh điển (Mario, Zelda, Pokémon). Mô hình đã học được rằng game mang nhãn Nintendo có xu hướng bán chạy hơn đáng kể.
    \item \textbf{Platform\_NES và Platform\_GB ($|\beta| \approx 1.13 - 1.32$):} Các nền tảng cổ điển này có doanh thu trung bình cao trong dataset do chúng gắn liền với thời kỳ Nintendo thống trị thị trường (1985-1995). Tuy nhiên, cần lưu ý đây một phần phản ánh hiệu ứng của \texttt{Publisher\_Nintendo} hơn là đặc trưng riêng của platform.
    \item \textbf{User\_Count xuất hiện trong top 10 ($|\beta| \approx 0.30$):} Số lượng người dùng đánh giá là proxy tốt cho mức độ phổ biến của game, có tương quan với doanh thu. Biến này nắm bắt thông tin về quy mô cộng đồng người chơi mà \texttt{Critic\_Score} không thể hiện được.
    \item \textbf{Genre\_Platform ($|\beta| = 0.237$):} Thể loại Platform (game nhảy/chạy theo màn) có hệ số dương, phù hợp với thực tế rằng nhiều tựa game bom tấn của Nintendo (Mario series) thuộc thể loại này.
\end{itemize}

\subsection{Kỹ Thuật Nâng Cao: Kernel Ridge Regression}

\subsubsection{Động lực nghiên cứu}
OLS và Ridge Regression giả định mối quan hệ tuyến tính toàn cục giữa đặc trưng và biến mục tiêu. Tuy nhiên, dữ liệu doanh thu game có thể chứa các phi tuyến tính cục bộ --- ví dụ, mối quan hệ giữa \texttt{Critic\_Score} và \texttt{Global\_Sales} có thể tăng tốc đột ngột ở vùng điểm rất cao (90-99). Kernel Ridge Regression được triển khai để kiểm chứng giả thuyết này và bổ sung một mô hình phi tham số vào bộ so sánh.

\subsubsection{Cơ sở lý thuyết}
Kernel Ridge Regression là phương pháp mở rộng OLS sang không gian đặc trưng phi tuyến thông qua kernel trick, theo đúng công thức đề bài mục 2.4:
\begin{equation}
    \hat{y}(\mathbf{x}) = \mathbf{k}(\mathbf{x})^T (\mathbf{K} + \lambda\mathbf{I})^{-1}\mathbf{y}
\end{equation}
Trong đó $\mathbf{K}$ là ma trận Gram với $K_{ij} = k(\mathbf{x}_i, \mathbf{x}_j)$, và $k(\cdot,\cdot)$ là hàm kernel RBF (Radial Basis Function):
\begin{equation}
    k(\mathbf{x}, \mathbf{x}') = \exp\left(-\frac{\|\mathbf{x} - \mathbf{x}'\|^2}{2\ell^2}\right)
\end{equation}
Tham số $\lambda$ đóng vai trò regularization (tương tự Ridge), còn $\ell$ (length scale) kiểm soát độ rộng của vùng ảnh hưởng của mỗi điểm train. Cả hai tham số được chọn đồng thời qua 3-fold Cross-Validation.

\subsubsection{Cài đặt}
Thuật toán được cài đặt từ đầu bằng Python thuần và NumPy, không sử dụng thư viện sklearn hay scikit-learn, đảm bảo tính minh bạch và phù hợp yêu cầu đề bài. Các bước chính bao gồm:
\begin{enumerate}
    \item Tính ma trận Gram $\mathbf{K}$ bằng vectorized NumPy.
    \item Giải hệ tuyến tính $(\mathbf{K} + \lambda\mathbf{I})\boldsymbol{\alpha} = \mathbf{y}$ bằng \texttt{np.linalg.solve} thay vì nghịch đảo trực tiếp để đảm bảo số trị ổn định.
    \item Dự đoán: $\hat{y} = \mathbf{K}_{\text{test}} \boldsymbol{\alpha}$.
\end{enumerate}

Do chi phí tính toán $\mathcal{O}(n^2)$ khi tính ma trận Gram, Kernel Ridge Regression được huấn luyện trên tập con 300 mẫu train và đánh giá trên 100 mẫu test. Chỉ 4 biến số (\texttt{Critic\_Score}, \texttt{Critic\_Count}, \texttt{User\_Score}, \texttt{User\_Count}) được sử dụng để tránh ``Curse of Dimensionality'' --- khi số chiều quá lớn, khoảng cách Euclidean mất khả năng phân biệt, khiến kernel weight gần như bằng nhau cho mọi cặp điểm.

Class \texttt{KernelRidgeRegression} trong file \texttt{part2/advanced\_methods.py} cài đặt đầy đủ logic này.

\subsubsection{Kết quả}
\begin{figure}[H]
    \centering
    \includegraphics[width=0.9\textwidth]{images_Kien_Phan5/kernel_regression_comparison.png}
    \caption{Kernel Ridge vs OLS --- ảnh hưởng của length scale $\ell$ (trái) và so sánh trực tiếp (phải).}
    \label{fig:kernel_comparison}
\end{figure}

Qua quá trình CV, cặp tham số tối ưu được chọn là $\lambda = 1.0$, $\ell = 1.0$. Kernel Ridge Regression đạt $\text{MAE} = 0.4612$, $\text{RMSE} = 0.8645$, $R^2 = 0.1441$ --- cải thiện nhẹ so với Ridge tuyến tính trên tập con này.

Biểu đồ so sánh trực quan (Hình \ref{fig:kernel_comparison}) minh họa rõ sự khác biệt về cơ chế giữa hai phương pháp: đường dự đoán của Kernel uốn lượn theo cấu trúc cục bộ của dữ liệu, trong khi OLS chỉ có thể fit một đường thẳng toàn cục. Biểu đồ trái cho thấy ảnh hưởng của $\ell$: $\ell$ nhỏ (0.3) gây overfit --- đường dự đoán dao động mạnh theo từng điểm, trong khi $\ell$ lớn (5.0) gây underfit --- đường dự đoán phẳng gần như một hằng số.

\subsubsection{Hạn chế}
\begin{itemize}
    \item \textbf{Chi phí tính toán $\mathcal{O}(n^2)$:} Không khả thi với toàn bộ 13,000+ mẫu train. Giải pháp mở rộng: Nyström approximation hoặc Random Fourier Features.
    \item \textbf{Không có hệ số diễn giải:} Kernel Ridge cho ra vector $\boldsymbol{\alpha}$ thay vì $\boldsymbol{\beta}$, không thể diễn giải trực tiếp tầm quan trọng đặc trưng.
    \item \textbf{Nhạy cảm với curse of dimensionality:} Phải giới hạn số features đầu vào, làm giảm thông tin so với Ridge tuyến tính dùng đủ 79 đặc trưng.
\end{itemize}

\subsection{Nhận Xét và Kết Luận Phần 2}

\begin{itemize}
    \item Bộ dữ liệu Video Games Sales có tỷ lệ missing values rất cao (trên 40\% ở nhiều cột), đòi hỏi pipeline tiền xử lý chặt chẽ.
    \item Ba mô hình OLS Full, OLS Stepwise và Ridge cho kết quả $R^2$ tương đương nhau trên tập test, cho thấy doanh thu game phụ thuộc chủ yếu vào một số ít đặc trưng cốt lõi.
    \item Phân tích phần dư cho thấy phân phối doanh thu lệch phải (right-skewed) và phương sai không đồng nhất --- đặc điểm phổ biến của dữ liệu doanh thu. Biến đổi log có thể cải thiện mô hình.
    \item Kernel Ridge Regression cho phép mô hình hóa mối quan hệ phi tuyến, nhưng trên tập dữ liệu thực với nhiều biến dummy, hiệu quả cải thiện không vượt trội so với Ridge tuyến tính.
    \item Các biến \texttt{Critic\_Count} và \texttt{Critic\_Score} là những đặc trưng quan trọng nhất trong việc dự báo doanh thu game.
\end{itemize}

\begin{enumerate}
    \item OLS tuy đơn giản nhưng vẫn là công cụ mạnh mẽ khi các giả thiết Gauss--Markov được thỏa mãn. Tuy nhiên, dữ liệu thực tế hiếm khi hoàn hảo --- cần luôn kiểm tra phần dư.
    \item Xử lý dữ liệu khuyết cần thận trọng: phải tách tập Train/Test trước khi impute để tránh data leakage.
    \item Ridge/Lasso regularization giúp ổn định mô hình khi đa cộng tuyến, nhưng không thay đổi bản chất nếu mối quan hệ giữa biến đầu vào và đầu ra là yếu.
    \item Kernel methods mở rộng khả năng mô hình hóa phi tuyến nhưng đánh đổi bằng chi phí tính toán $O(n^3)$.
\end{enumerate}

\end{itemize}
